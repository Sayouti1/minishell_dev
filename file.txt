#include <unistd.h>
#include <fcntl.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 1024

int read_and_write(const char *filename) {
    int fd = open(filename, O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) {
        perror("Error opening file");
        return 1;
    }

    char buffer[BUFFER_SIZE];
    ssize_t bytes_read;

    while ((bytes_read = read(STDIN_FILENO, buffer, BUFFER_SIZE)) > 0) {
        if (bytes_read >= 3 && strncmp(buffer, "END", 3) == 0) {
            break;
        }

        if (write(fd, buffer, bytes_read) != bytes_read) {
            perror("Write error");
            close(fd);
            return 1;
        }
    }

    if (bytes_read == -1) {
        perror("Read error");
        close(fd);
        return 1;
    }

    close(fd);
    return 0;
}
